cmake_minimum_required(VERSION 3.16)
project(algorithm_review LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
foreach(name IN ITEMS 01_foundations 02_linear 03_hash_window 04_sort_search 05_tree 06_heap_dsu 07_graph 08_shortest_path 09_backtracking 10_greedy 11_dp 12_string_range)
  add_executable(${name} examples/${name}.cpp)
  target_include_directories(${name} PRIVATE tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME ${name} COMMAND ${name})
endforeach()

# 学习者模板：显式指定目标才会编译，不计入参考示例的 CTest。
foreach(number IN ITEMS 01 02 03 04 05 06 07 08 09 10 11 12)
  add_executable(practice_${number} EXCLUDE_FROM_ALL exercises/${number}_practice.cpp)
  target_include_directories(practice_${number} PRIVATE tests)
endforeach()
