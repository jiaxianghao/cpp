cmake_minimum_required(VERSION 3.16)
project(cpp_review LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(REVIEW_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
enable_testing()
function(review_example name source)
  add_executable(${name} ${source})
  target_include_directories(${name} PRIVATE ${PROJECT_SOURCE_DIR}/include)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
    if(REVIEW_SANITIZERS)
      target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${name} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
  add_test(NAME ${name} COMMAND ${name})
endfunction()
review_example(01_lifetime examples/01_lifetime.cpp)
review_example(02_ownership examples/02_ownership.cpp)
review_example(03_copy_move examples/03_copy_move.cpp)
review_example(04_object_model examples/04_object_model.cpp)
review_example(05_exceptions examples/05_exceptions.cpp)
review_example(06_templates examples/06_templates.cpp)
review_example(07_stl examples/07_stl.cpp)
review_example(08_refactoring examples/08_refactoring.cpp)
review_example(09_patterns examples/09_patterns.cpp)
review_example(10_checkout project/checkout.cpp)
